# pypx MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a modern PyPI frontend (pypx.app) with enriched package data, fast search, and terminal-hybrid aesthetics.

**Architecture:** Nuxt 4 SSR frontend + Go caching/enrichment API behind Caddy, deployed via Docker Compose on DigitalOcean with Cloudflare in front. Go fetches from PyPI JSON API and pypistats.org, enriches data (install sizes, dep trees, compatibility), caches in SQLite, and exposes a REST API that Nuxt consumes.

**Tech Stack:** Go 1.23+ (chi, modernc.org/sqlite), Nuxt 4 (SSR, Nuxt UI, Tailwind 4, VueUse, Vitest), Caddy, Docker Compose

**Spec:** `docs/superpowers/specs/2026-04-09-pypx-design.md`

```go
package main

import (
	"context"
	"log"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/go-chi/chi/v5"
	"github.com/go-chi/chi/v5/middleware"
	"github.com/pypx/api/internal/handler"
)

func main() {
	r := chi.NewRouter()
	r.Use(middleware.Logger)
	r.Use(middleware.Recoverer)
	r.Use(middleware.Timeout(30 * time.Second))

	r.Get("/api/health", handler.Health)

	port := os.Getenv("API_PORT")
	if port == "" {
		port = "8080"
	}

	srv := &http.Server{
		Addr:    ":" + port,
		Handler: r,
	}

	go func() {
		log.Printf("pypx api listening on :%s", port)
		if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("server error: %v", err)
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit

	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	if err := srv.Shutdown(ctx); err != nil {
		log.Fatalf("shutdown error: %v", err)
	}
	log.Println("server stopped")
}
```

```go
package cache

import (
	"database/sql"
	"time"

	_ "modernc.org/sqlite"
)

type Cache struct {
	db *sql.DB
}

func New(dsn string) (*Cache, error) {
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}

	db.Exec("PRAGMA journal_mode=WAL")
	db.Exec("PRAGMA synchronous=NORMAL")

	_, err = db.Exec(`
		CREATE TABLE IF NOT EXISTS cache (
			key TEXT PRIMARY KEY,
			value BLOB NOT NULL,
			created_at INTEGER NOT NULL
		)
	`)
	if err != nil {
		db.Close()
		return nil, err
	}

	return &Cache{db: db}, nil
}

func (c *Cache) Set(key string, value []byte, ttl time.Duration) error {
	_, err := c.db.Exec(
		`INSERT OR REPLACE INTO cache (key, value, created_at) VALUES (?, ?, ?)`,
		key, value, time.Now().UnixMilli(),
	)
	return err
}

func (c *Cache) Get(key string, ttl time.Duration) (data []byte, fresh bool, err error) {
	var value []byte
	var createdAt int64

	err = c.db.QueryRow(
		`SELECT value, created_at FROM cache WHERE key = ?`, key,
	).Scan(&value, &createdAt)

	if err == sql.ErrNoRows {
		return nil, false, nil
	}
	if err != nil {
		return nil, false, err
	}

	age := time.Since(time.UnixMilli(createdAt))
	return value, age < ttl, nil
}

func (c *Cache) Close() error {
	return c.db.Close()
}
```

Create `api/internal/pypi/client.go`:

```go
package pypi

import (
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

type PackageInfo struct {
	Name            string            `json:"name"`
	Version         string            `json:"version"`
	Summary         string            `json:"summary"`
	Description     string            `json:"description"`
	DescriptionType string            `json:"description_content_type"`
	License         string            `json:"license"`
	Author          string            `json:"author"`
	AuthorEmail     string            `json:"author_email"`
	HomePage        string            `json:"home_page"`
	RequiresPython  string            `json:"requires_python"`
	RequiresDist    []string          `json:"requires_dist"`
	ProjectURLs     map[string]string `json:"project_urls"`
	Classifiers     []string          `json:"classifiers"`
	Keywords        string            `json:"keywords"`
}

type ReleaseFile struct {
	Filename    string `json:"filename"`
	URL         string `json:"url"`
	Size        int64  `json:"size"`
	PackageType string `json:"packagetype"`
	PythonVer   string `json:"python_version"`
	RequiresPy  string `json:"requires_python"`
	UploadTime  string `json:"upload_time_iso_8601"`
	Digests     struct {
		SHA256 string `json:"sha256"`
	} `json:"digests"`
}

type PyPIResponse struct {
	Info     PackageInfo                `json:"info"`
	Releases map[string][]ReleaseFile  `json:"releases"`
	URLs     []ReleaseFile             `json:"urls"`
}

type Client struct {
	baseURL    string
	httpClient *http.Client
}

type Option func(*Client)

func WithBaseURL(url string) Option {
	return func(c *Client) {
		c.baseURL = url
	}
}

func NewClient(opts ...Option) *Client {
	c := &Client{
		baseURL: "https://pypi.org",
		httpClient: &http.Client{
			Timeout: 15 * time.Second,
		},
	}
	for _, opt := range opts {
		opt(c)
	}
	return c
}

func (c *Client) FetchPackage(name string) (*PyPIResponse, error) {
	url := fmt.Sprintf("%s/pypi/%s/json", c.baseURL, name)

	resp, err := c.httpClient.Get(url)
	if err != nil {

	var result PyPIResponse
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {

	}

	return &result, nil
```

```go
func (h *PackageHandler) Get(w http.ResponseWriter, r *http.Request) {
	name := chi.URLParam(r, "name")
	cacheKey := "pkg:" + strings.ToLower(name)

	// Check cache
	data, fresh, err := h.cache.Get(cacheKey, packageTTL)
	if err == nil && data != nil && fresh {
		w.Header().Set("Content-Type", "application/json")
		w.Write(data)
		return
	}

	// Stale cache: return stale data but we'd queue refresh in production
	// For now, fetch fresh if stale or missing
	if data != nil && !fresh {
		w.Header().Set("Content-Type", "application/json")
		w.Write(data)
		// TODO: queue background refresh (Task 27)
		return
	}

	// Cache miss: fetch from PyPI
	pkg, err := h.pypi.FetchPackage(name)
	if err != nil {
		if strings.Contains(err.Error(), "not found") {
			http.Error(w, `{"error":"package not found"}`, http.StatusNotFound)
			return
		}

		return
```

```go
package enrichment

import (
	"strings"

	"github.com/pypx/api/internal/pypi"
)

type PythonVersionInfo struct {
	Constraint string `json:"constraint"`
	MinVersion string `json:"min_version"`
}

// ExtractInstallSize returns the size of the smallest wheel, falling back to sdist.
func ExtractInstallSize(files []pypi.ReleaseFile) int64 {
	var wheelSize, sdistSize int64

	for _, f := range files {
		switch f.PackageType {
		case "bdist_wheel":
			if wheelSize == 0 || f.Size < wheelSize {
				wheelSize = f.Size
			}
		case "sdist":
			if sdistSize == 0 || f.Size < sdistSize {
				sdistSize = f.Size
			}
		}
	}

	if wheelSize > 0 {
		return wheelSize
	}
	return sdistSize
}

// ExtractModuleFormat determines if the package is pure Python or platform-specific.
func ExtractModuleFormat(files []pypi.ReleaseFile) string {
	for _, f := range files {
		if f.PackageType != "bdist_wheel" {
			continue
		}
		name := f.Filename
		// Wheel filename: {name}-{ver}-{python}-{abi}-{platform}.whl
		parts := strings.Split(strings.TrimSuffix(name, ".whl"), "-")
		if len(parts) >= 5 {
			platform := parts[len(parts)-1]
			abi := parts[len(parts)-2]
			python := parts[len(parts)-3]
			tag := python + "-" + abi + "-" + platform
			if platform == "any" && abi == "none" {
				return tag
			}
			return "platform-specific"
		}
	}
	return "sdist-only"
}

// ExtractPythonVersions parses the requires_python specifier.
func ExtractPythonVersions(requiresPython string) PythonVersionInfo {
	info := PythonVersionInfo{
		Constraint: requiresPython,
	}

	if requiresPython == "" {
		return info
	}

	// Extract minimum version from specifiers like ">=3.8" or ">=3.9,<4.0"
	parts := strings.Split(requiresPython, ",")
	for _, part := range parts {
		part = strings.TrimSpace(part)
		if strings.HasPrefix(part, ">=") {
			info.MinVersion = strings.TrimPrefix(part, ">=")
			break
		}
		if strings.HasPrefix(part, ">") {
			info.MinVersion = strings.TrimPrefix(part, ">")
			break
		}
	}

	return info
}
```

```go
package enrichment

import (
	"regexp"
	"strings"
)

type Dependency struct {
	Name       string `json:"name"`
	Constraint string `json:"constraint"`
}

type DependencyTree struct {
	Required []Dependency            `json:"required"`
	Extras   map[string][]Dependency `json:"extras"`
}

var extraPattern = regexp.MustCompile(`extra\s*==\s*"([^"]+)"`)

func ParseDependencies(requiresDist []string) DependencyTree {
	tree := DependencyTree{
		Required: []Dependency{},
		Extras:   map[string][]Dependency{},
	}

	for _, spec := range requiresDist {
		dep, extra := parseSingleDep(spec)

		if extra != "" {
			tree.Extras[extra] = append(tree.Extras[extra], dep)
		} else {
			tree.Required = append(tree.Required, dep)
		}
	}

	return tree
}

func parseSingleDep(spec string) (Dependency, string) {
	var extra string

	// Split on semicolon to separate markers
	parts := strings.SplitN(spec, ";", 2)
	depPart := strings.TrimSpace(parts[0])

	if len(parts) == 2 {
		marker := strings.TrimSpace(parts[1])
		if match := extraPattern.FindStringSubmatch(marker); len(match) == 2 {
			extra = match[1]
		} else {
			// Non-extra environment marker (e.g., platform_system == "Windows")
			// Treat as a required dep for MVP
			extra = ""
		}
	}

	// Parse name and version constraint
	dep := Dependency{}
	if idx := strings.Index(depPart, " ("); idx != -1 {
		dep.Name = strings.TrimSpace(depPart[:idx])
		constraint := depPart[idx+2:]
		constraint = strings.TrimSuffix(constraint, ")")
		dep.Constraint = strings.TrimSpace(constraint)
	} else if idx := strings.Index(depPart, "("); idx != -1 {
		dep.Name = strings.TrimSpace(depPart[:idx])
		constraint := depPart[idx+1:]
		constraint = strings.TrimSuffix(constraint, ")")
		dep.Constraint = strings.TrimSpace(constraint)
	} else {
		dep.Name = strings.TrimSpace(depPart)
	}

	return dep, extra
}
```

```go
func buildVersionsList(pkg *pypi.PyPIResponse) []VersionInfo {
	var versions []VersionInfo

	for ver, files := range pkg.Releases {
		if len(files) == 0 {
			continue
		}

		pypiFiles := make([]pypi.ReleaseFile, len(files))
		copy(pypiFiles, files)

		vi := VersionInfo{
			Version:      ver,
			InstallSize:  enrichment.ExtractInstallSize(pypiFiles),
			ModuleFormat: enrichment.ExtractModuleFormat(pypiFiles),
		}

		for _, f := range files {
			if f.UploadTime != "" && vi.UploadTime == "" {
				vi.UploadTime = f.UploadTime
			}
			vi.Files = append(vi.Files, FileInfo{
				Filename:    f.Filename,
				Size:        f.Size,
				PackageType: f.PackageType,
				PythonVer:   f.PythonVer,
				UploadTime:  f.UploadTime,
			})
		}

		versions = append(versions, vi)
	}

	return versions
}
```

```go
package stats

import (
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

type DataPoint struct {
	Category  string `json:"category"`
	Downloads int64  `json:"downloads"`
}

type StatsResponse struct {
	Data    []DataPoint `json:"data"`
	Package string      `json:"package"`
	Type    string      `json:"type"`
}

type Client struct {
	baseURL    string
	httpClient *http.Client
}

type Option func(*Client)

func WithBaseURL(url string) Option {
	return func(c *Client) {
		c.baseURL = url
	}
}

func NewClient(opts ...Option) *Client {
	c := &Client{
		baseURL: "https://pypistats.org",
		httpClient: &http.Client{
			Timeout: 15 * time.Second,
		},
	}
	for _, opt := range opts {
		opt(c)
	}
	return c
}

func (c *Client) fetch(url string) (*StatsResponse, error) {
	resp, err := c.httpClient.Get(url)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("unexpected status %d", resp.StatusCode)
	}

	var result StatsResponse
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return nil, err
	}
	return &result, nil
}

func (c *Client) FetchOverall(name string) (*StatsResponse, error) {
	return c.fetch(fmt.Sprintf("%s/api/packages/%s/overall", c.baseURL, name))
}

func (c *Client) FetchPythonVersions(name string) (*StatsResponse, error) {
	return c.fetch(fmt.Sprintf("%s/api/packages/%s/python_minor", c.baseURL, name))
}

func (c *Client) FetchSystem(name string) (*StatsResponse, error) {
	return c.fetch(fmt.Sprintf("%s/api/packages/%s/system", c.baseURL, name))
}
```

```go
	Downloads int64  `json:"downloads"`
}

type Index struct {
	db *sql.DB
}

func NewIndex(dsn string) (*Index, error) {
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}

	_, err = db.Exec(`
		CREATE VIRTUAL TABLE IF NOT EXISTS packages_fts USING fts5(
			name,
			summary,
			downloads UNINDEXED,
			tokenize='porter unicode61'
		)
	`)
	if err != nil {
		db.Close()
		return nil, fmt.Errorf("creating FTS table: %w", err)
	}

	// Auxiliary table for exact-match and popularity ranking
	_, err = db.Exec(`
		CREATE TABLE IF NOT EXISTS packages_meta (
			name TEXT PRIMARY KEY,
			summary TEXT,
			downloads INTEGER DEFAULT 0
		)
	`)
	if err != nil {
		db.Close()
		return nil, err
	}

	return &Index{db: db}, nil
}

func (idx *Index) Upsert(entry PackageEntry) error {
	tx, err := idx.db.Begin()
	if err != nil {
		return err
	}
	defer tx.Rollback()

	// Upsert meta table
	_, err = tx.Exec(
		`INSERT OR REPLACE INTO packages_meta (name, summary, downloads) VALUES (?, ?, ?)`,
		entry.Name, entry.Summary, entry.Downloads,
	)
	if err != nil {
		return err
	}

	// Delete old FTS entry and reinsert
	tx.Exec(`DELETE FROM packages_fts WHERE name = ?`, entry.Name)
	_, err = tx.Exec(
		`INSERT INTO packages_fts (name, summary, downloads) VALUES (?, ?, ?)`,
		entry.Name, entry.Summary, entry.Downloads,
	)
	if err != nil {
		return err
	}

	return tx.Commit()
}

func (idx *Index) Search(query string, limit int) ([]PackageEntry, error) {
	if strings.TrimSpace(query) == "" {
		return nil, nil
	}

	// Prefix matching: append * for typeahead
	ftsQuery := strings.ReplaceAll(query, `"`, `""`)
	ftsQuery = `"` + ftsQuery + `"*`

	rows, err := idx.db.Query(`
		SELECT m.name, m.summary, m.downloads
		FROM packages_fts f
		JOIN packages_meta m ON f.name = m.name
		WHERE packages_fts MATCH ?
		ORDER BY
			CASE WHEN m.name = ? THEN 0 ELSE 1 END,
			m.downloads DESC
		LIMIT ?
	`, ftsQuery, query, limit)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var results []PackageEntry
	for rows.Next() {
		var entry PackageEntry
		if err := rows.Scan(&entry.Name, &entry.Summary, &entry.Downloads); err != nil {
			return nil, err
		}
		results = append(results, entry)
	}
	return results, rows.Err()
}

func (idx *Index) Close() error {
	return idx.db.Close()
}
```

```go
type Config struct {
	SimpleAPIURL    string
	IndexSyncEvery  time.Duration
	StaleCheckEvery time.Duration
}

type Worker struct {
	pypi   *pypi.Client
	cache  *cache.Cache
	index  *search.Index
	config Config
}

func New(pypiClient *pypi.Client, c *cache.Cache, idx *search.Index, cfg Config) *Worker {
	if cfg.SimpleAPIURL == "" {
		cfg.SimpleAPIURL = "https://pypi.org/simple/"
	}
	if cfg.IndexSyncEvery == 0 {
		cfg.IndexSyncEvery = 6 * time.Hour
	}
	if cfg.StaleCheckEvery == 0 {
		cfg.StaleCheckEvery = 5 * time.Minute
	}
	return &Worker{pypi: pypiClient, cache: c, index: idx, config: cfg}
}

var linkPattern = regexp.MustCompile(`<a[^>]+href="/simple/([^/]+)/"`)

// SyncIndex fetches the PyPI Simple API package list and indexes package names.
// For MVP, it indexes names only. Metadata enrichment happens on first access.
func (w *Worker) SyncIndex(ctx context.Context) error {
	log.Println("worker: starting index sync")

	resp, err := http.Get(w.config.SimpleAPIURL)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}

	matches := linkPattern.FindAllStringSubmatch(string(body), -1)
	log.Printf("worker: found %d packages in Simple API", len(matches))

	count := 0
	for _, match := range matches {
		if ctx.Err() != nil {
			break
		}
		if len(match) < 2 {
			continue
		}
		name := strings.TrimSpace(match[1])
		w.index.Upsert(search.PackageEntry{
			Name:    name,
			Summary: "",
		})
		count++
	}

	log.Printf("worker: indexed %d packages", count)
	return nil
}

// Start launches background workers. Call with a cancellable context.
func (w *Worker) Start(ctx context.Context) {
	// Initial sync
	go func() {
		if err := w.SyncIndex(ctx); err != nil {
			log.Printf("worker: initial index sync failed: %v", err)
		}
	}()

	// Periodic re-sync
	go func() {
		ticker := time.NewTicker(w.config.IndexSyncEvery)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				if err := w.SyncIndex(ctx); err != nil {
					log.Printf("worker: periodic index sync failed: %v", err)
				}
			}
		}
	}()
}
```

```vue
      query.value = ''
      results.value = []
      selectedIndex.value = 0
    }
  }
  if (e.key === 'Escape') {
    isOpen.value = false
  }
})

// Debounced search
const debouncedSearch = useDebounceFn(async (q: string) => {
  if (!q.trim()) {
    results.value = []
    return
  }
  results.value = await searchPackages(q, 8)
  selectedIndex.value = 0
}, 150)

watch(query, (val) => debouncedSearch(val))

function navigate(name: string) {
  isOpen.value = false
  router.push(`/packages/${name}`)
}

function onKeyDown(e: KeyboardEvent) {
  if (e.key === 'ArrowDown') {
    e.preventDefault()
    selectedIndex.value = Math.min(selectedIndex.value + 1, results.value.length - 1)
  } else if (e.key === 'ArrowUp') {
    e.preventDefault()
    selectedIndex.value = Math.max(selectedIndex.value - 1, 0)
  } else if (e.key === 'Enter' && results.value[selectedIndex.value]) {
    navigate(results.value[selectedIndex.value].name)
  }
}

function formatDownloads(n: number): string {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
  if (n >= 1_000) return `${(n / 1_000).toFixed(0)}K`
  return String(n)
}
</script>

      leave-active-class="transition-opacity duration-75"

              :class="i === selectedIndex ? 'bg-zinc-800 text-zinc-50' : 'text-zinc-400 hover:bg-zinc-800/50'"
              @click="navigate(result.name)"
              @mouseenter="selectedIndex = i"
            >

                <span class="font-mono text-sm">{{ result.name }}</span>

                </span>
              </div>
              <span v-if="result.downloads" class="font-mono text-xs text-zinc-600">

              </span>
```
